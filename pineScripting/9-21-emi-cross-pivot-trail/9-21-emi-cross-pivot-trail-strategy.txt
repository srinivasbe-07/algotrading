//@version=6
strategy("9-21 EMA Crossover-pivot-trail", overlay=true, default_qty_type=strategy.fixed, pyramiding=0)

// Inputs
emaFastLength = input.int(9, "Fast EMA Length")
emaSlowLength = input.int(21, "Slow EMA Length")
use_atr_distance = input.bool(title="Use ATR for Distance Filter", defval=true)
distanceMultiplier = input.float(title="Distance Multiplier (ATR)", defval=0.5, step=0.1)
minBarsBetweenCrosses = input.int(5, "Min Bars Between Crossovers", minval=1)
enableReentries = input.bool(true, "Enable Reentries")
maxReentries = input.int(2, "Max Reentries", minval=0)
pivotBarsLeft = input.int(5, "Pivot Left Bars", minval=1)
pivotBarsRight = input.int(5, "Pivot Right Bars", minval=1)

// Trade Tracking
var int crossBarIndex = 0
var float entryPrice = na
var bool isLong = false
var bool isShort = false
var int reentryCount = 0
var bool lastExitWasLong = false
var bool lastExitWasShort = false
var int entryBar = na
var float longStopPrice = na
var float shortStopPrice = na

// Position Check Functions
noPosition() => strategy.position_size == 0
isPositionLong() => strategy.position_size > 0
isPositionShort() => strategy.position_size < 0

// Define the ticker symbol
symbol = syminfo.ticker

// Set quantities based on the symbol
quantity = switch symbol
    'ETHUSD.P' => 20
    'XRPUSD.P' => 120
    'DOGEUSD.P' => 15
    'BTCUSD.P' => 4
    'SOLUSD.P' => 3

if symbol == 'BTCUSD.P'
    maxReentries := 3
    minBarsBetweenCrosses := 3
    
// Calculate EMAs
emaFast = ta.ema(close, emaFastLength)
emaSlow = ta.ema(close, emaSlowLength)

// Crossover Conditions
bullishCross = ta.crossover(emaFast, emaSlow)
bearishCross = ta.crossunder(emaFast, emaSlow)

// Reentry Conditions
highestHigh5 = ta.highest(high, 5)[1]
lowestLow5 = ta.lowest(low, 5)[1]

// ATR for distance filter
atr = ta.atr(14)
minDistance = use_atr_distance ? (atr * distanceMultiplier) : na

// Pivot Detection
pivotLow() => ta.pivotlow(low, pivotBarsLeft, pivotBarsRight)
pivotHigh() => ta.pivothigh(high, pivotBarsLeft, pivotBarsRight)

// Update stops when new pivots are detected
if not na(pivotLow()) and isLong
    longStopPrice := low[pivotBarsRight]

if not na(pivotHigh()) and isShort
    shortStopPrice := high[pivotBarsRight]

// Exit Conditions
longExitCondition = (low <= longStopPrice) or bearishCross
shortExitCondition = (high >= shortStopPrice) or bullishCross

// Exit Orders - close any existing position first
if isPositionLong() and longExitCondition
    strategy.close("Long", comment = "Exit Long")
    isLong := false
    entryPrice := na
    entryBar := na
    lastExitWasLong := true
    lastExitWasShort := false
    longStopPrice := na

if isPositionShort() and shortExitCondition
    strategy.close("Short", comment = "Exit Short")
    isShort := false
    entryPrice := na
    entryBar := na
    lastExitWasShort := true
    lastExitWasLong := false
    shortStopPrice := na

// Entry Conditions with ATR Distance Filter
longDistanceOk = not use_atr_distance ? true : (math.abs(close - emaFast) > minDistance) and minBarsBetweenCrosses < bar_index - crossBarIndex
shortDistanceOk = not use_atr_distance ? true : (math.abs(close - emaSlow) > minDistance) and minBarsBetweenCrosses < bar_index - crossBarIndex

// Strategy Entry - only enter if no position exists
if noPosition() and bullishCross and longDistanceOk
    recentPivot = low[pivotBarsRight]
    strategy.entry("Long", strategy.long, qty=quantity, comment="Long")
    isLong := true
    entryPrice := close
    entryBar := bar_index
    reentryCount := 0
    lastExitWasLong := false
    lastExitWasShort := false
    longStopPrice := recentPivot

if noPosition() and bearishCross and shortDistanceOk
    recentPivot = high[pivotBarsRight]
    strategy.entry("Short", strategy.short, qty=quantity, comment="Short")
    isShort := true
    entryPrice := close
    entryBar := bar_index
    reentryCount := 0
    lastExitWasLong := false
    lastExitWasShort := false
    shortStopPrice := recentPivot

// Strategy Logic - Reentries
longReentry = enableReentries and noPosition() and lastExitWasLong and close > highestHigh5 and reentryCount < maxReentries
if longReentry
    recentPivot = low[pivotBarsRight]
    strategy.entry("Long", strategy.long, qty=quantity, comment="Reentry Long")
    isLong := true
    entryPrice := close
    entryBar := bar_index
    reentryCount := reentryCount + 1
    lastExitWasLong := false
    lastExitWasShort := false
    longStopPrice := recentPivot

shortReentry = enableReentries and noPosition() and lastExitWasShort and close < lowestLow5 and reentryCount < maxReentries
if shortReentry
    recentPivot = high[pivotBarsRight]
    strategy.entry("Short", strategy.short, qty=quantity, comment="Reentry Short")
    isShort := true
    entryPrice := close
    entryBar := bar_index
    reentryCount := reentryCount + 1
    lastExitWasLong := false
    lastExitWasShort := false
    shortStopPrice := recentPivot

if bullishCross or bearishCross
    crossBarIndex := bar_index
    lastExitWasShort := false
    lastExitWasLong := false

if bullishCross and not isLong
    lastExitWasLong := true
if bearishCross and not isShort
    lastExitWasShort := true
    
// Plotting
plot(emaFast, color=color.new(color.blue, 0), title="9 EMA")
plot(emaSlow, color=color.new(color.red, 0), title="21 EMA")

// Highlight Crossovers
plotshape(bullishCross, style=shape.xcross, location=location.belowbar, color=color.new(color.green, 0), size=size.tiny, title="Bullish Cross")
plotshape(bearishCross, style=shape.xcross, location=location.abovebar, color=color.new(color.red, 0), size=size.tiny, title="Bearish Cross")

// Plot pivot stops

plot(isLong  ? longStopPrice : na, "Long Trail", color.new(color.green, 50), 2, plot.style_linebr)
plot(isShort  ? shortStopPrice : na, "Short Trail", color.new(color.red, 50), 2, plot.style_linebr)

//@version=6
strategy("Enhanced EMA Crossover w/ Smart Re-Entry", overlay=true, default_qty_type=strategy.fixed, pyramiding=0)

// Inputs
emaFastLength = input.int(9, "Fast EMA Length")
emaSlowLength = input.int(21, "Slow EMA Length")
trailActive = input.bool(true, "Activate Trailing Stop")
baseTrailMultiplier = input.float(0.3, "Base Trail Multiplier", minval=0.1, step=0.1)

// Progressive Trail Settings
enableDynamicTrail = input.bool(true, "Enable Progressive Trail")
progressionBars = input.int(5, "Bars Between Progression", minval=1)
maxMultiplier = input.float(0.75, "Max Trail Multiplier", minval=0.1, step=0.1)

// Re-entry Settings
reEntryEnabled = input.bool(true, "Enable Smart Re-Entry")
resumptionBars = input.int(1000, "Re-entry Window Bars", minval=1)
minPriceRecovery = input.float(0.5, "Min Price Recovery (%)", minval=0.1, step=0.1) / 100
volumeFactor = input.float(1.5, "Volume Multiplier", minval=1.0, step=0.1)

// Define the ticker symbol
symbol = syminfo.ticker

// Set quantities based on the symbol
quantity = switch symbol
    'ETHUSD.P' => 20
    'XRPUSD.P1' => 120
    'DOGEUSD.P1' => 15
    'BTCUSD.P' => 4
    'SOLUSD.P' => 3

// Calculate EMAs
emaFast = ta.ema(close, emaFastLength)
emaSlow = ta.ema(close, emaSlowLength)

// Crossover Conditions
bullishCross = ta.crossover(emaFast, emaSlow)
bearishCross = ta.crossunder(emaFast, emaSlow)

// Trade Tracking
var float entryCandleSize = na
var float entryPrice = na
var float entryLow = na
var float entryHigh = na
var bool isLong = false
var bool isShort = false

// Re-entry tracking
var float lastExitPrice = na
var int lastExitBar = na
var string lastExitType = na // "long" or "short"

// Trailing Stop Variables
var float longTrailPrice = na
var float shortTrailPrice = na
var float longPeak = na
var float shortValley = na
var int entryBar = na

// Exit Conditions
longStopPrice = trailActive ? (close < longTrailPrice) : (close < entryLow)
shortStopPrice = trailActive ? (close > shortTrailPrice) : (close > entryHigh)

// Function to get current progressive multiplier
getCurrentMultiplier() =>
    if not enableDynamicTrail
        baseTrailMultiplier
    else
        barsSinceEntry = bar_index - entryBar
        steps = math.floor(barsSinceEntry / progressionBars)
        math.min(baseTrailMultiplier + (steps * 0.1), maxMultiplier)

// Exit Orders
if(isLong and (longStopPrice or bearishCross))
    strategy.close("Long")
    
    // Capture exit state for re-entry
    lastExitPrice := close
    lastExitBar := bar_index
    lastExitType := "long"
    
    isLong := false
    entryCandleSize := na
    entryPrice := na
    entryLow := na
    longPeak := na
    longTrailPrice := na
    entryBar := na

if(isShort and (shortStopPrice or bullishCross))
    strategy.close("Short")

    // Capture exit state for re-entry
    lastExitPrice := close
    lastExitBar := bar_index
    lastExitType := "short"
    
    isShort := false
    entryCandleSize := na
    entryPrice := na
    entryHigh := na
    shortValley := na
    shortTrailPrice := na
    entryBar := na

// SMART Re-entry Conditions =============================================

// 1. EMA Crossover Retrigger
crossoverReentryLong = bullishCross and (lastExitType == "long") and 
                      (bar_index - lastExitBar <= resumptionBars)

crossoverReentryShort = bearishCross and (lastExitType == "short") and 
                       (bar_index - lastExitBar <= resumptionBars)

// 2. Price Recovery Re-entry
priceRecoveryLong = (lastExitType == "long") and 
                 (close > lastExitPrice * (1 + minPriceRecovery)) and 
                 (emaFast > emaSlow) and 
                 (bar_index - lastExitBar <= resumptionBars)

priceRecoveryShort = (lastExitType == "short") and 
                 (close < lastExitPrice * (1 - minPriceRecovery)) and 
                 (emaFast < emaSlow) and 
                 (bar_index - lastExitBar <= resumptionBars)

// 3. Volume-Supported Re-entry
avgVolume = ta.sma(volume, 20)
volumeSpike = volume > avgVolume * volumeFactor

volumeReentryLong = (lastExitType == "long") and 
                   (close > lastExitPrice) and 
                   volumeSpike and 
                   (emaFast > emaSlow) and 
                   (bar_index - lastExitBar <= resumptionBars)

volumeReentryShort = (lastExitType == "short") and 
                 (close < lastExitPrice) and 
                 volumeSpike and 
                 (emaFast < emaSlow) and 
                 (bar_index - lastExitBar <= resumptionBars)

// Combined Re-entry Signals
longResumption = reEntryEnabled and not isLong and not isShort and 
             (crossoverReentryLong or priceRecoveryLong or volumeReentryLong)

shortResumption = reEntryEnabled and not isLong and not isShort and 
                 (crossoverReentryShort or priceRecoveryShort or volumeReentryShort)

// Strategy Entry Logic =================================================
if not isLong and not isShort and (bullishCross or longResumption)
    strategy.entry("Long", strategy.long, qty=quantity)
    
    // Reset re-entry tracking
    lastExitPrice := na
    lastExitBar := na
    lastExitType := na
    
    isLong := true
    entryPrice := close
    entryCandleSize := high - low
    entryLow := low
    longPeak := high
    longTrailPrice := low
    entryBar := bar_index

if not isLong and not isShort and (bearishCross or shortResumption)
    strategy.entry("Short", strategy.short, qty=quantity)
    
    // Reset re-entry tracking
    lastExitPrice := na
    lastExitBar := na
    lastExitType := na
    
    isShort := true
    entryPrice := close
    entryCandleSize := high - low
    entryHigh := high
    shortValley := low
    shortTrailPrice := high
    entryBar := bar_index

// Update Trailing Levels
if isLong and bar_index > entryBar
    longPeak := math.max(longPeak, high)
    currentMult = getCurrentMultiplier()
    longTrailPrice := entryLow + (longPeak - entryLow) * currentMult
    
if isShort and bar_index > entryBar
    shortValley := math.min(shortValley, low)
    currentMult = getCurrentMultiplier()
    shortTrailPrice := entryHigh - (entryHigh - shortValley) * currentMult

// Plotting
plot(emaFast, color=color.new(color.blue, 0), title="9 EMA")
plot(emaSlow, color=color.new(color.red, 0), title="21 EMA")

// Visualize Trailing Stops
plot(isLong and trailActive ? longTrailPrice : na, "Long Trail", color.new(color.yellow, 70), 2, plot.style_linebr)
plot(isShort and trailActive ? shortTrailPrice : na, "Short Trail", color.new(color.black, 70), 2, plot.style_linebr)

// Highlight Crossovers
plotshape(bullishCross, style=shape.triangleup, location=location.belowbar, color=color.new(color.green, 0), size=size.small, title="Bullish Cross")
plotshape(bearishCross, style=shape.triangledown, location=location.abovebar, color=color.new(color.red, 0), size=size.small, title="Bearish Cross")

// Visual Indicators ====================================================
// Re-entry signals
plotshape(longResumption and crossoverReentryLong, style=shape.labelup, location=location.belowbar, color=color.new(color.blue, 0), text="CROSS RE", size=size.small)
plotshape(longResumption and priceRecoveryLong, style=shape.labelup, location=location.belowbar, color=color.new(color.purple, 0), text="PRICE RE", size=size.small)
plotshape(longResumption and volumeReentryLong, style=shape.labelup, location=location.belowbar, color=color.new(color.orange, 0), text="VOL RE", size=size.small)

plotshape(shortResumption and crossoverReentryShort, style=shape.labeldown, location=location.abovebar, color=color.new(color.blue, 0), text="CROSS RE", size=size.small)
plotshape(shortResumption and priceRecoveryShort, style=shape.labeldown, location=location.abovebar, color=color.new(color.purple, 0), text="PRICE RE", size=size.small)
plotshape(shortResumption and volumeReentryShort, style=shape.labeldown, location=location.abovebar, color=color.new(color.orange, 0), text="VOL RE", size=size.small)
